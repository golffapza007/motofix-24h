    // Toggle Language Function (3A)
    function toggleLanguage() {
      currentLang = (currentLang === 'TH') ? 'EN' : 'TH';
      document.getElementById('currentLangLabel').innerText = (currentLang === 'TH') ? 'TH | EN' : 'EN | TH';
      
      // Update UI elements with data-i18n attribute
      document.querySelectorAll('[data-i18n]').forEach(elem => {
        const key = elem.getAttribute('data-i18n');
        if (translations[currentLang][key]) {
          elem.innerHTML = translations[currentLang][key];
        }
      });
    }

    // Get GPS Location Function
    function getGPSLocation() {
      const gpsInput = document.getElementById('gpsInput');
      if (navigator.geolocation) {
        gpsInput.value = (currentLang === 'TH') ? "กำลังดึงพิกัด..." : "Fetching coordinates...";
        navigator.geolocation.getCurrentPosition(
          (position) => {
            const lat = position.coords.latitude.toFixed(6);
            const lng = position.coords.longitude.toFixed(6);
            gpsInput.value = `Lat: ${lat}, Lng: ${lng}`;
          },
          (error) => {
            gpsInput.value = "Lat: 13.7563, Lng: 100.5018 (Bangkok)";
          }
        );
      } else {
        gpsInput.value = "Lat: 13.7563, Lng: 100.5018 (Bangkok)";
      }
    }

    // Search Repair Status Demo (2A)
    function searchRepairStatus() {
      const val = document.getElementById('trackInput').value.trim();
      const resultBox = document.getElementById('trackingResult');
      
      if (val === "") {
        alert(currentLang === 'TH' ? "กรุณากรอกเบอร์โทรหรือรหัสใบรับรถ" : "Please enter phone number or ticket code");
        return;
      }
      
      resultBox.classList.remove('hidden');
      if (val === "MOTO-102" || val === "0812345678") {
        document.getElementById('ticketCode').innerText = "MOTO-102";
        document.getElementById('bikeModel').innerText = "Honda Click 160 (กก-1234)";
        document.getElementById('estCompletion').innerText = (currentLang === 'TH') ? "วันนี้ 18:30 น." : "Today 18:30";
      } else {
        document.getElementById('ticketCode').innerText = val.toUpperCase();
        document.getElementById('bikeModel').innerText = "Yamaha Aerox 155 (ขข-5678)";
        document.getElementById('estCompletion').innerText = (currentLang === 'TH') ? "พรุ่งนี้ 10:00 น." : "Tomorrow 10:00";
      }
    }
